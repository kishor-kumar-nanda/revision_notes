from typing import TypedDict, Annotated, Literal
from langchain_core.messages import BaseMessage, HumanMessage, ToolMessage
from langchain_core.tools import tool
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from langchain_openai import ChatOpenAI

@tool
def query_latency(service: str) -> str:
    """Queries p99 latency for a service."""
    return f"{service} p99 latency: 42ms"

tools_by_name = {"query_latency": query_latency}
llm = ChatOpenAI(model="gpt-4o-mini").bind_tools([query_latency])

class ReActState(TypedDict):
    messages: Annotated[list[BaseMessage], add_messages]

def call_agent(state: ReActState):
    return {"messages": [llm.invoke(state["messages"])]}

def execute_tools(state: ReActState):
    last_msg = state["messages"][-1]
    tool_results = []
    for call in last_msg.tool_calls:
        tool_fn = tools_by_name[call["name"]]
        output = tool_fn.invoke(call["args"])
        tool_results.append(ToolMessage(content=str(output), tool_call_id=call["id"]))
    return {"messages": tool_results}

def route_decision(state: ReActState) -> Literal["tools", "__end__"]:
    last_msg = state["messages"][-1]
    return "tools" if last_msg.tool_calls else END

workflow = StateGraph(ReActState)
workflow.add_node("agent", call_agent)
workflow.add_node("tools", execute_tools)

workflow.add_edge(START, "agent")
workflow.add_conditional_edges("agent", route_decision)
workflow.add_edge("tools", "agent")

app = workflow.compile()
print("Custom ReAct state machine ready for execution.")

class DeployState(TypedDict):

def plan_deploy(state: DeployState):

from pydantic import BaseModel
from langchain_openai import ChatOpenAI

class TeamState(TypedDict):
    messages: Annotated[list[BaseMessage], add_messages]
    next_worker: str

class RoutingDecision(BaseModel):
    next: Literal["coder", "tester", "FINISH"]

def supervisor_node(state: TeamState):

builder.add_node("coder", coder_worker)
builder.add_node("tester", tester_worker)

builder.add_conditional_edges("supervisor", lambda s: s["next_worker"], {

    "tester": "tester",
    "FINISH": END

from pydantic import BaseModel
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI

class PlanSchema(BaseModel):
    steps: list[str]

class PlanState(TypedDict):
    objective: str
    todos: list[dict]
    current_step: int

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

def planner_node(state: PlanState):
    planner = llm.with_structured_output(PlanSchema)
    plan = planner.invoke(f"Decompose this task into steps: {state['objective']}")
    todos = [{"task": step, "done": False} for step in plan.steps]
    return {"todos": todos, "current_step": 0}

def step_executor_node(state: PlanState):
    idx = state["current_step"]
    todos = list(state["todos"])
    todos[idx]["done"] = True  # Mark current step complete
    return {"todos": todos, "current_step": idx + 1}

def routing_condition(state: PlanState) -> Literal["execute", "__end__"]:
    return "execute" if state["current_step"] < len(state["todos"]) else END

builder = StateGraph(PlanState)
builder.add_node("plan", planner_node)
builder.add_node("execute", step_executor_node)

builder.add_edge(START, "plan")
builder.add_edge("plan", "execute")
builder.add_conditional_edges("execute", routing_condition)

app = builder.compile()

class GuardState(TypedDict):
    query: str
    is_safe: bool
    response: str

def input_guardrail_node(state: GuardState):
    query_lower = state["query"].lower()
    is_safe = not any(bad in query_lower for bad in FORBIDDEN_KEYWORDS)
    return {"is_safe": is_safe}

    return {"response": "SECURITY POLICY VIOLATION: Request blocked by input guardrail."}

    return {"response": "Executed successfully."}

builder.add_node("process", processing_node)
builder.add_node("reject", rejection_node)
